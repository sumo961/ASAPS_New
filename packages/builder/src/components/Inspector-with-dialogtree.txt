import React from 'react';
import { Beat } from '@asaps/core';
import { X, Save, Trash2, Copy, Info, Plus, Link, Unlink, MapPin, Package, Settings, AlertCircle, MessageSquare } from 'lucide-react';
import beatDefinitions from '../../../../beat-definitions/core-beats.json';
import { DialogTreeEditor } from '../editors/DialogTreeEditor';

/**
 * ENHANCED INSPECTOR WITH DIALOG TREE SUPPORT
 * 
 * Features:
 * 1. Full DialogTree editor integration
 * 2. Proper connection display (no duplicates)
 * 3. Counter connections for choices
 * 4. Improved parameter persistence
 * 5. Better multi-connection beat handling
 * 6. Visual dialog tree editing
 */

interface InspectorProps {
  beat: Beat | null;
  onUpdate: (beatId: string, updates: Partial<Beat>) => void;
  onDelete: (beatId: string) => void;
  allBeats?: Beat[];
  onConnect?: (sourceBeatId: string, targetBeatId: string) => void;
  onDisconnect?: (sourceBeatId: string, targetBeatId: string) => void;
}

interface ChoiceWithCounter {
  id: string;
  text: string;
  location?: string;
  target: string;
  counterEffect?: {
    counter: string;
    operation: 'add' | 'subtract' | 'set';
    value: number;
  };
}

interface PropWithEffect {
  id: string;
  name: string;
  description: string;
  target: string;
  effect?: {
    type: 'inventory' | 'variable';
    name: string;
    value: any;
  };
}

export const Inspector: React.FC<InspectorProps> = ({
  beat,
  onUpdate,
  onDelete,
  allBeats = [],
  onConnect,
  onDisconnect,
}) => {
  const [localBeat, setLocalBeat] = React.useState<any>(null);
  const [hasChanges, setHasChanges] = React.useState(false);
  const [showAdvanced, setShowAdvanced] = React.useState(false);
  const [validationErrors, setValidationErrors] = React.useState<string[]>([]);

  // Get beat definition from schema
  const getBeatDefinition = (beatType: string) => {
    return beatDefinitions.beatTypes[beatType as keyof typeof beatDefinitions.beatTypes];
  };

  React.useEffect(() => {
    if (beat) {
      // Create a deep copy of the beat data including parameters
      const beatData = beat.toJSON();
      
      // Get unique connections (no duplicates)
      const connections = beat.getConnections ? beat.getConnections() : [];
      const uniqueConnections = Array.from(
        new Map(connections.map(c => [`${c.targetId}-${c.label}`, c])).values()
      );
      beatData.connections = uniqueConnections;
      
      // Ensure parameters object exists
      beatData.parameters = beat.getParameters ? beat.getParameters() : {};
      
      // Initialize data for specific beat types
      const beatDef = getBeatDefinition(beat.type);
      if (beatDef?.connectionType === 'multiple') {
        if (beat.type === 'movementChoice' && !beatData.parameters.choices) {
          beatData.parameters.choices = [];
        } else if (beat.type === 'pickProp' && !beatData.parameters.props) {
          beatData.parameters.props = [];
        } else if (beat.type === 'dialogTree' && !beatData.parameters.dialogTree) {
          beatData.parameters.dialogTree = {
            id: 'root',
            speaker: beatData.parameters.speaker || 'Character',
            text: beatData.parameters.text || 'Hello...',
            emotion: beatData.parameters.emotion || 'neutral'
          };
        }
      }
      
      setLocalBeat(beatData);
      setHasChanges(false);
      setValidationErrors([]);
    }
  }, [beat?.id, beat?.name]); // Only re-run when beat ID or name changes

  if (!beat || !localBeat) {
    return (
      <div className="w-80 bg-white border-l border-gray-200 p-6">
        <div className="text-center text-gray-500">
          <Info className="w-12 h-12 mx-auto mb-3 text-gray-300" />
          <p className="text-sm">Select a beat to view its properties</p>
        </div>
      </div>
    );
  }

  const beatDef = getBeatDefinition(beat.type);
  const connectionType = beatDef?.connectionType || 'single';

  const validateBeat = (): string[] => {
    const errors: string[] = [];
    
    // Validate required fields based on beat type
    switch (beat.type) {
      case 'titleScreen':
        if (!localBeat.parameters?.title) errors.push('Title is required');
        break;
      case 'introText':
        if (!localBeat.parameters?.text) errors.push('Text content is required');
        break;
      case 'dialogTree':
        if (!localBeat.parameters?.dialogTree) {
          errors.push('Dialog tree is required');
        } else {
          if (!localBeat.parameters.dialogTree.speaker) errors.push('Speaker is required');
          if (!localBeat.parameters.dialogTree.text) errors.push('Dialog text is required');
        }
        break;
      case 'movementChoice':
        if (!localBeat.parameters?.choices?.length) {
          errors.push('At least one choice is required');
        } else {
          localBeat.parameters.choices.forEach((choice: any, i: number) => {
            if (!choice.target) errors.push(`Choice ${i + 1} needs a target`);
          });
        }
        break;
      case 'pickProp':
        if (!localBeat.parameters?.props?.length) {
          errors.push('At least one prop is required');
        } else {
          localBeat.parameters.props.forEach((prop: any, i: number) => {
            if (!prop.target) errors.push(`Prop ${i + 1} needs a target`);
          });
        }
        break;
      case 'conditionBeat':
        if (!localBeat.parameters?.condition) errors.push('Condition is required');
        const conns = localBeat.connections || [];
        const hasTrue = conns.some((c: any) => c.label === 'true');
        const hasFalse = conns.some((c: any) => c.label === 'false');
        if (!hasTrue) errors.push('True connection is required');
        break;
    }
    
    // Validate single-connection beats have a connection
    if (connectionType === 'single' && beat.type !== 'endScreen' && beat.type !== 'dialogTree') {
      if (!localBeat.connections?.length) {
        errors.push('Connection is required');
      }
    }
    
    return errors;
  };

  const handleChange = (field: string, value: any) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      [field]: value,
    }));
    setHasChanges(true);
  };

  const handleParameterChange = (param: string, value: any) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        [param]: value,
      },
    }));
    setHasChanges(true);
  };

  const handleDialogTreeChange = (tree: any) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        dialogTree: tree,
        // Also update root speaker/text for backward compatibility
        speaker: tree.speaker,
        text: tree.text,
        emotion: tree.emotion
      },
    }));
    setHasChanges(true);
  };

  const handleSave = () => {
    // Validate before saving
    const errors = validateBeat();
    if (errors.length > 0) {
      setValidationErrors(errors);
      return;
    }
    
    if (beat) {
      // Update basic properties
      beat.name = localBeat.name;
      beat.cluster = localBeat.cluster;
      beat.defaultTarget = localBeat.defaultTarget || undefined;
      beat.transition = localBeat.transition;
      beat.sound = localBeat.sound;
      
      // Update parameters
      if (localBeat.parameters && beat.updateParameters) {
        beat.updateParameters(localBeat.parameters);
      }
      
      // Clear ALL existing connections first
      const beatAny = beat as any;
      if (typeof beatAny.clearConnections === 'function') {
        beatAny.clearConnections();
      } else {
        // Direct assignment for older API
        beat.connections = [];
      }
      
      // Re-add connections based on type
      if (beat.type === 'dialogTree') {
        // Extract connections from dialog tree choices
        const extractConnections = (node: any) => {
          const connections: any[] = [];
          
          if (node.choices) {
            node.choices.forEach((choice: any) => {
              if (typeof choice.target === 'string' && choice.target) {
                connections.push({
                  targetId: choice.target,
                  label: choice.text
                });
              } else if (typeof choice.target === 'object' && choice.target) {
                // Recursively extract from nested nodes
                connections.push(...extractConnections(choice.target));
              }
            });
          }
          
          if (typeof node.next === 'string' && node.next) {
            connections.push({
              targetId: node.next,
              label: 'Continue'
            });
          } else if (typeof node.next === 'object' && node.next) {
            connections.push(...extractConnections(node.next));
          }
          
          return connections;
        };
        
        const dialogConnections = extractConnections(localBeat.parameters.dialogTree);
        
        // Also add any default connection after dialog ends
        if (localBeat.connections?.length > 0) {
          const defaultConn = localBeat.connections.find((c: any) => !c.label || c.label === 'Continue');
          if (defaultConn) {
            dialogConnections.push(defaultConn);
          }
        }
        
        // Add unique connections
        const uniqueConns = Array.from(
          new Map(dialogConnections.map(c => [`${c.targetId}-${c.label}`, c])).values()
        );
        uniqueConns.forEach(conn => beat.addConnection(conn));
        
      } else if (connectionType === 'single' && localBeat.connections.length > 0) {
        beat.addConnection(localBeat.connections[0]);
      } else if (connectionType === 'multiple') {
        // For multiple connection beats, connections come from choices/props
        if (beat.type === 'movementChoice' && localBeat.parameters?.choices) {
          localBeat.parameters.choices.forEach((choice: ChoiceWithCounter) => {
            if (choice.target) {
              beat.addConnection({
                targetId: choice.target,
                label: choice.text
              });
            }
          });
        } else if (beat.type === 'pickProp' && localBeat.parameters?.props) {
          localBeat.parameters.props.forEach((prop: PropWithEffect) => {
            if (prop.target) {
              beat.addConnection({
                targetId: prop.target,
                label: prop.name
              });
            }
          });
        }
      } else if (connectionType === 'conditional') {
        // Handle conditional beats
        localBeat.connections.forEach((conn: any) => beat.addConnection(conn));
      } else {
        // Add all connections for other types
        localBeat.connections.forEach((conn: any) => beat.addConnection(conn));
      }
      
      // Trigger update
      onUpdate(beat.id, beat);
      setHasChanges(false);
      setValidationErrors([]);
    }
  };

  const handleDelete = () => {
    if (window.confirm(`Are you sure you want to delete "${beat.name}"?`)) {
      onDelete(beat.id);
    }
  };

  // Handle adding choices for movementChoice beats with counter effects
  const handleAddChoice = () => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        choices: [
          ...(prev.parameters.choices || []),
          {
            id: `choice_${Date.now()}`,
            text: 'New Choice',
            location: 'Location',
            target: '',
            counterEffect: undefined
          }
        ]
      }
    }));
    setHasChanges(true);
  };

  const handleUpdateChoice = (index: number, field: string, value: any) => {
    setLocalBeat((prev: any) => {
      const choices = [...(prev.parameters.choices || [])];
      if (field.startsWith('counterEffect.')) {
        const subField = field.split('.')[1];
        choices[index] = {
          ...choices[index],
          counterEffect: {
            ...choices[index].counterEffect,
            [subField]: value
          }
        };
      } else {
        choices[index] = { ...choices[index], [field]: value };
      }
      return {
        ...prev,
        parameters: {
          ...prev.parameters,
          choices
        }
      };
    });
    setHasChanges(true);
  };

  const handleRemoveChoice = (index: number) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        choices: prev.parameters.choices.filter((_: any, i: number) => i !== index)
      }
    }));
    setHasChanges(true);
  };

  // Handle adding props for pickProp beats with effects
  const handleAddProp = () => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        props: [
          ...(prev.parameters.props || []),
          {
            id: `prop_${Date.now()}`,
            name: 'New Prop',
            description: 'Description',
            target: '',
            effect: undefined
          }
        ]
      }
    }));
    setHasChanges(true);
  };

  const handleUpdateProp = (index: number, field: string, value: any) => {
    setLocalBeat((prev: any) => {
      const props = [...(prev.parameters.props || [])];
      if (field.startsWith('effect.')) {
        const subField = field.split('.')[1];
        props[index] = {
          ...props[index],
          effect: {
            ...props[index].effect,
            [subField]: value
          }
        };
      } else {
        props[index] = { ...props[index], [field]: value };
      }
      return {
        ...prev,
        parameters: {
          ...prev.parameters,
          props
        }
      };
    });
    setHasChanges(true);
  };

  const handleRemoveProp = (index: number) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      parameters: {
        ...prev.parameters,
        props: prev.parameters.props.filter((_: any, i: number) => i !== index)
      }
    }));
    setHasChanges(true);
  };

  const handleAddConnection = (targetId: string, label?: string) => {
    if (!targetId) return;

    if (connectionType === 'single') {
      // Replace existing connection for single-connection beats
      setLocalBeat((prev: any) => ({
        ...prev,
        connections: [{
          targetId: targetId,
          label: label || prev.parameters?.buttonText || 'Continue'
        }]
      }));
    } else if (connectionType === 'conditional') {
      // Add as true/false connection
      const hasTrue = localBeat.connections.some((c: any) => c.label === 'true');
      setLocalBeat((prev: any) => ({
        ...prev,
        connections: [
          ...prev.connections.filter((c: any) => c.label !== (hasTrue ? 'false' : 'true')),
          {
            targetId: targetId,
            label: hasTrue ? 'false' : 'true'
          }
        ]
      }));
    }
    
    setHasChanges(true);
  };

  const handleRemoveConnection = (targetId: string, label?: string) => {
    setLocalBeat((prev: any) => ({
      ...prev,
      connections: prev.connections.filter((c: any) => 
        !(c.targetId === targetId && (!label || c.label === label))
      )
    }));
    setHasChanges(true);
  };

  // Get available characters for dialog tree
  const getAvailableCharacters = () => {
    // TODO: Pull from story settings/characters
    return ['Player', 'NPC', 'Narrator', 'Guide', 'Merchant', 'Guard'];
  };

  const availableTargets = allBeats.filter(b => b.id !== beat.id);

  return (
    <div className="w-80 bg-white border-l border-gray-200 flex flex-col">
      {/* Header */}
      <div className="p-4 border-b border-gray-200">
        <div className="flex items-center justify-between mb-2">
          <h2 className="text-lg font-semibold text-gray-800">Properties</h2>
          {hasChanges && (
            <span className="text-xs text-orange-500 font-medium">Unsaved changes</span>
          )}
        </div>
        <div className="text-sm text-gray-500">
          {beat.type} • ID: {beat.id}
        </div>
        <div className="text-xs text-gray-400 mt-1">
          Connection Type: {connectionType}
        </div>
      </div>

      {/* Validation Errors */}
      {validationErrors.length > 0 && (
        <div className="p-3 bg-red-50 border-b border-red-200">
          <div className="flex items-start gap-2">
            <AlertCircle className="w-4 h-4 text-red-500 mt-0.5" />
            <div className="text-sm text-red-700">
              <div className="font-medium mb-1">Please fix the following:</div>
              {validationErrors.map((error, i) => (
                <div key={i} className="text-xs">• {error}</div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* Properties */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {/* Basic Properties */}
        <div className="space-y-3">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Name
            </label>
            <input
              type="text"
              value={localBeat.name}
              onChange={(e) => handleChange('name', e.target.value)}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
          </div>

          {/* Beat-specific Parameters */}
          {beat.type === 'titleScreen' && (
            <>
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">Title</label>
                <input
                  type="text"
                  value={localBeat.parameters?.title || ''}
                  onChange={(e) => handleParameterChange('title', e.target.value)}
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm"
                />
              </div>
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">Author</label>
                <input
                  type="text"
                  value={localBeat.parameters?.author || ''}
                  onChange={(e) => handleParameterChange('author', e.target.value)}
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm"
                />
              </div>
            </>
          )}

          {beat.type === 'introText' && (
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">Text Content</label>
              <textarea
                value={localBeat.parameters?.text || ''}
                onChange={(e) => handleParameterChange('text', e.target.value)}
                rows={4}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm"
              />
            </div>
          )}

          {/* DIALOG TREE EDITOR */}
          {beat.type === 'dialogTree' && (
            <div className="space-y-3">
              <DialogTreeEditor
                dialogTree={localBeat.parameters?.dialogTree || {
                  id: 'root',
                  speaker: 'Character',
                  text: 'Hello...',
                  emotion: 'neutral'
                }}
                onChange={handleDialogTreeChange}
                characters={getAvailableCharacters()}
                allBeats={availableTargets}
              />
              
              {/* Default connection after dialog ends */}
              <div className="border-t pt-3">
                <h3 className="text-sm font-medium text-gray-700 mb-2">
                  After Dialog Ends
                </h3>
                <select
                  value={localBeat.connections?.[0]?.targetId || ''}
                  onChange={(e) => {
                    if (e.target.value) {
                      handleAddConnection(e.target.value, 'Continue');
                    } else {
                      setLocalBeat((prev: any) => ({
                        ...prev,
                        connections: []
                      }));
                      setHasChanges(true);
                    }
                  }}
                  className="w-full px-2 py-1 border border-gray-300 rounded text-sm"
                >
                  <option value="">End story here</option>
                  {availableTargets.map(target => (
                    <option key={target.id} value={target.id}>
                      {target.name} ({target.id})
                    </option>
                  ))}
                </select>
              </div>
            </div>
          )}

          {beat.type === 'movementChoice' && (
            <div className="space-y-3">
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">Question</label>
                <input
                  type="text"
                  value={localBeat.parameters?.question || 'Where do you want to go?'}
                  onChange={(e) => handleParameterChange('question', e.target.value)}
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm"
                />
              </div>
              
              <div>
                <div className="flex items-center justify-between mb-2">
                  <span className="text-sm font-medium text-gray-700">Choices</span>
                  <button
                    onClick={handleAddChoice}
                    className="text-xs bg-green-500 text-white px-2 py-1 rounded hover:bg-green-600"
                  >
                    <Plus className="w-3 h-3 inline" /> Add
                  </button>
                </div>
                
                {localBeat.parameters?.choices?.map((choice: ChoiceWithCounter, index: number) => (
                  <div key={choice.id} className="p-3 bg-gray-50 rounded-lg space-y-2 mb-2">
                    <div className="flex justify-between">
                      <span className="text-xs font-medium">Choice {index + 1}</span>
                      <button
                        onClick={() => handleRemoveChoice(index)}
                        className="text-red-500 hover:bg-red-50 p-1 rounded"
                      >
                        <X className="w-3 h-3" />
                      </button>
                    </div>
                    
                    <input
                      type="text"
                      value={choice.text}
                      onChange={(e) => handleUpdateChoice(index, 'text', e.target.value)}
                      placeholder="Choice text"
                      className="w-full px-2 py-1 text-sm border rounded"
                    />
                    
                    <input
                      type="text"
                      value={choice.location || ''}
                      onChange={(e) => handleUpdateChoice(index, 'location', e.target.value)}
                      placeholder="Location description"
                      className="w-full px-2 py-1 text-sm border rounded"
                    />
                    
                    <select
                      value={choice.target || ''}
                      onChange={(e) => handleUpdateChoice(index, 'target', e.target.value)}
                      className="w-full px-2 py-1 text-sm border rounded"
                    >
                      <option value="">Select target beat...</option>
                      {availableTargets.map(target => (
                        <option key={target.id} value={target.id}>
                          {target.name} ({target.id})
                        </option>
                      ))}
                    </select>
                    
                    {/* Counter Effect (Advanced) */}
                    {showAdvanced && (
                      <div className="p-2 bg-blue-50 rounded space-y-2">
                        <div className="text-xs font-medium text-blue-700">Counter Effect (Optional)</div>
                        <input
                          type="text"
                          value={choice.counterEffect?.counter || ''}
                          onChange={(e) => handleUpdateChoice(index, 'counterEffect.counter', e.target.value)}
                          placeholder="Counter name (e.g., courage)"
                          className="w-full px-2 py-1 text-xs border rounded"
                        />
                        <div className="flex gap-2">
                          <select
                            value={choice.counterEffect?.operation || 'add'}
                            onChange={(e) => handleUpdateChoice(index, 'counterEffect.operation', e.target.value)}
                            className="flex-1 px-2 py-1 text-xs border rounded"
                          >
                            <option value="add">Add</option>
                            <option value="subtract">Subtract</option>
                            <option value="set">Set</option>
                          </select>
                          <input
                            type="number"
                            value={choice.counterEffect?.value || 0}
                            onChange={(e) => handleUpdateChoice(index, 'counterEffect.value', parseInt(e.target.value))}
                            placeholder="Value"
                            className="w-20 px-2 py-1 text-xs border rounded"
                          />
                        </div>
                      </div>
                    )}
                  </div>
                ))}
              </div>
            </div>
          )}

          {beat.type === 'pickProp' && (
            <div className="space-y-3">
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">Question</label>
                <input
                  type="text"
                  value={localBeat.parameters?.question || 'What do you want to pick up?'}
                  onChange={(e) => handleParameterChange('question', e.target.value)}
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg text-sm"
                />
              </div>
              
              <div>
                <div className="flex items-center justify-between mb-2">
                  <span className="text-sm font-medium text-gray-700">Props</span>
                  <button
                    onClick={handleAddProp}
                    className="text-xs bg-green-500 text-white px-2 py-1 rounded hover:bg-green-600"
                  >
                    <Plus className="w-3 h-3 inline" /> Add
                  </button>
                </div>
                
                {localBeat.parameters?.props?.map((prop: PropWithEffect, index: number) => (
                  <div key={prop.id} className="p-3 bg-gray-50 rounded-lg space-y-2 mb-2">
                    <div className="flex justify-between">
                      <span className="text-xs font-medium">Prop {index + 1}</span>
                      <button
                        onClick={() => handleRemoveProp(index)}
                        className="text-red-500 hover:bg-red-50 p-1 rounded"
                      >
                        <X className="w-3 h-3" />
                      </button>
                    </div>
                    
                    <input
                      type="text"
                      value={prop.name}
                      onChange={(e) => handleUpdateProp(index, 'name', e.target.value)}
                      placeholder="Prop name"
                      className="w-full px-2 py-1 text-sm border rounded"
                    />
                    
                    <input
                      type="text"
                      value={prop.description}
                      onChange={(e) => handleUpdateProp(index, 'description', e.target.value)}
                      placeholder="Description"
                      className="w-full px-2 py-1 text-sm border rounded"
                    />
                    
                    <select
                      value={prop.target || ''}
                      onChange={(e) => handleUpdateProp(index, 'target', e.target.value)}
                      className="w-full px-2 py-1 text-sm border rounded"
                    >
                      <option value="">Select target beat...</option>
                      {availableTargets.map(target => (
                        <option key={target.id} value={target.id}>
                          {target.name} ({target.id})
                        </option>
                      ))}
                    </select>
                    
                    {/* Inventory Effect (Advanced) */}
                    {showAdvanced && (
                      <div className="p-2 bg-green-50 rounded space-y-2">
                        <div className="text-xs font-medium text-green-700">Inventory Effect (Optional)</div>
                        <select
                          value={prop.effect?.type || 'inventory'}
                          onChange={(e) => handleUpdateProp(index, 'effect.type', e.target.value)}
                          className="w-full px-2 py-1 text-xs border rounded"
                        >
                          <option value="inventory">Add to Inventory</option>
                          <option value="variable">Set Variable</option>
                        </select>
                        <input
                          type="text"
                          value={prop.effect?.name || prop.name}
                          onChange={(e) => handleUpdateProp(index, 'effect.name', e.target.value)}
                          placeholder="Item/Variable name"
                          className="w-full px-2 py-1 text-xs border rounded"
                        />
                      </div>
                    )}
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* Connections for single/conditional beats (NOT dialog tree) */}
          {(connectionType === 'single' || connectionType === 'conditional') && beat.type !== 'dialogTree' && (
            <div className="border-t pt-3">
              <h3 className="text-sm font-medium text-gray-700 mb-2">
                Connections
              </h3>
              
              {localBeat.connections?.map((conn: any) => {
                const targetBeat = allBeats.find(b => b.id === conn.targetId);
                return (
                  <div key={`${conn.targetId}-${conn.label}`} className="flex items-center gap-2 p-2 bg-gray-50 rounded mb-2">
                    {connectionType === 'conditional' && (
                      <span className={`text-xs font-medium px-2 py-1 rounded ${
                        conn.label === 'true' ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'
                      }`}>
                        {conn.label?.toUpperCase()}
                      </span>
                    )}
                    <span className="flex-1 text-sm">
                      → {targetBeat?.name || conn.targetId}
                    </span>
                    <button
                      onClick={() => handleRemoveConnection(conn.targetId, conn.label)}
                      className="p-1 text-red-500 hover:bg-red-50 rounded"
                    >
                      <Unlink className="w-3 h-3" />
                    </button>
                  </div>
                );
              })}
              
              {/* Add connection UI */}
              {(connectionType === 'single' && !localBeat.connections?.length) ||
               (connectionType === 'conditional' && localBeat.connections?.length < 2) ? (
                <div className="flex gap-2">
                  <select
                    onChange={(e) => handleAddConnection(e.target.value)}
                    className="flex-1 px-2 py-1 border border-gray-300 rounded text-sm"
                    value=""
                  >
                    <option value="">Add connection...</option>
                    {availableTargets.map(target => (
                      <option key={target.id} value={target.id}>
                        {target.name} ({target.id})
                      </option>
                    ))}
                  </select>
                </div>
              ) : null}
            </div>
          )}
        </div>
      </div>

      {/* Footer Actions */}
      <div className="p-4 border-t border-gray-200 space-y-2">
        <div className="flex gap-2 mb-2">
          <button
            onClick={() => setShowAdvanced(!showAdvanced)}
            className="flex-1 py-2 px-4 border border-gray-300 rounded-lg text-sm font-medium text-gray-700 hover:bg-gray-50 transition-colors flex items-center justify-center gap-2"
          >
            <Settings className="w-4 h-4" />
            {showAdvanced ? 'Hide' : 'Show'} Advanced
          </button>
        </div>
        
        <button
          onClick={handleSave}
          disabled={!hasChanges}
          className={`w-full py-2 px-4 rounded-lg text-sm font-medium transition-colors flex items-center justify-center gap-2 ${
            hasChanges
              ? 'bg-blue-500 text-white hover:bg-blue-600'
              : 'bg-gray-100 text-gray-400 cursor-not-allowed'
          }`}
        >
          <Save className="w-4 h-4" />
          Save Changes
        </button>
        
        <button
          onClick={handleDelete}
          className="w-full py-2 px-4 border border-red-300 rounded-lg text-sm font-medium text-red-600 hover:bg-red-50 transition-colors flex items-center justify-center gap-2"
        >
          <Trash2 className="w-4 h-4" />
          Delete Beat
        </button>
      </div>
    </div>
  );
};
